# TreasureIQ API dependencies.
#
# Lower bounds rather than exact pins: this is an application people are meant
# to clone and run, and hard pins would break on any environment whose wheels
# differ. The lock for reproducible deploys belongs in the container image.

fastapi>=0.141,<1.0
uvicorn[standard]>=0.52,<1.0
pydantic>=2.13,<3.0
httpx>=0.28,<1.0
itsdangerous>=2.2,<3.0

# Optional. The extraction cache in data/extraction-cache/ is committed, so the
# application runs end to end without this package and without an API key.
# Install it only to re-run extraction against sources that changed.
anthropic>=0.120,<1.0

# Imported lazily by ingest/wp_pages.py, only when a candidate page links a
# PDF (D-15). Without it, PDF attachments are skipped and logged rather than
# failing the run — install it to recover requirements locked in bandi PDFs.
pypdf>=5.1,<6.0
snowballstemmer>=3.0

# Optional. Loaded lazily by chat/filtri.py (ciclo 11, D-06) to scope
# negations in free text ("non sono disabile") via lemma + dependency
# structure. Without it, filtri.py falls back to a deterministic cue-list
# and degrades without raising (L-5) — so the app and its tests run fine
# with just this line, no model installed.
#
# The 500MB `it_core_news_lg` model is deliberately NOT pinned here: this
# file is `pip install -r`'d by the base image too (see Dockerfile), which
# stays minimal on purpose for the citizen-facing container. filtri.py IS
# wired into production (B3): the base image runs on the cue-list fallback
# above, an honest degradation (L-5), not the spaCy model. The model is
# fetched by a dedicated Dockerfile step instead, pinned to the exact spaCy-compatible
# release (spaCy models are versioned per spaCy release; an unpinned model
# would silently pick whatever matches at install time).
spacy>=3.8,<3.9
